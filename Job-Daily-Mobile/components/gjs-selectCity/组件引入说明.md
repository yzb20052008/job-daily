组件示例：

![组件示例](https://img-blog.csdnimg.cn/0c4abd088cbb43649f714586ca90fcbd.gif)

页面代码：

```html
	<gjsSelectCity :scrollHeight="scrollHeight" :selectedColor="selectedColor" @select="select"></gjsSelectCity>
```

引入组件：

```js
import gjsSelectCity from '@/components/gjs-selectCity/gjs-selectCity.vue'
export default {
	components: {
		gjsSelectCity
	},
	data() {
		return {
			scrollHeight: 800,//scroll-view的高度
			selectedColor:"#55aaff",//选中的字体颜色
		}
	},
	onLoad() {
		this.scrollHeight = uni.getSystemInfoSync().windowHeight * 2
	},
	methods: {
		select(e) {
			console.log('选中的地区', e);
		}
	}
}
```
